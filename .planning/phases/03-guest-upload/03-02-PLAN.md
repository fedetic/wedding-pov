```typescript
import { db } from '@/lib/db';
import { events } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import { notFound } from 'next/navigation';
import { GuestUploadClient } from '@/components/guest/GuestUploadClient';

export const dynamic = 'force-dynamic'; // never cache — event.isActive can change at any time

interface Props {
  params: Promise<{ slug: string }>;
}

export async function generateMetadata({ params }: Props) {
  const { slug } = await params;
  const [event] = await db
    .select({ name: events.name })
    .from(events)
    .where(eq(events.slug, slug))
    .limit(1);
  return {
    title: event ? `${event.name} — Upload Photos` : 'Upload Photos',
  };
}

export default async function GuestUploadPage({ params }: Props) {
  const { slug } = await params;

  const [event] = await db
    .select({
      id: events.id,
      name: events.name,
      slug: events.slug,
      photoLimit: events.photoLimit,
      isActive: events.isActive,
      driveFolderId: events.driveFolderId,
    })
    .from(events)
    .where(eq(events.slug, slug))
    .limit(1);

  // Not found — render 404-style error (do not call notFound() — show friendly message instead)
  if (!event) {
    return (
      <main className="min-h-screen flex flex-col items-center justify-center bg-[#faf9f7]">
        <div className="w-full max-w-sm px-6 py-12 text-center">
          <h1 className="text-3xl font-semibold text-gray-800 mt-8">

          </h1>
          <p className="text-base text-gray-500 mt-3">
            Double-check the QR code and try scanning again.
          </p>
        </div>
      </main>
    );
  }

  // Inactive — render inactive error (server-side, no JS required)
  if (!event.isActive) {
    return (
      <main className="min-h-screen flex flex-col items-center justify-center bg-[#faf9f7]">
        <div className="w-full max-w-sm px-6 py-12 text-center">
          <h1 className="text-3xl font-semibold text-gray-800 mt-8">

          </h1>
          <p className="text-base text-gray-500 mt-3">
            Check with the event organizer.
          </p>
        </div>
      </main>
    );
  }

  // Active — hand off to client state machine
  // Pass only serializable primitives — no DB row objects with Date fields
  return (
    <GuestUploadClient
      event={{
        id: event.id,
        name: event.name,
        slug: event.slug,
        photoLimit: event.photoLimit,
```

**State machine steps:**
```
STEP_NICKNAME → STEP_SELECT → STEP_UPLOADING → STEP_COMPLETE
                                    └──(any fail)──→ ERROR_UPLOAD
```

```typescript
interface PhotoState {

  status: PhotoStatus;
  progress: number; // 0–100
}
```

**Full component — implement exactly as specified:**

```tsx
'use client';

import { useRef, useState, useEffect } from 'react';

type Step = 'nickname' | 'select' | 'uploading' | 'complete' | 'error_upload';

type PhotoStatus = 'pending' | 'uploading' | 'done' | 'failed';

interface PhotoState {
  file: File;
  status: PhotoStatus;
  progress: number; // 0–100
}

interface EventProps {
  id: string;
  name: string;
  slug: string;
  photoLimit: number;
}

// ── helpers ──────────────────────────────────────────────────────────────────

function plural(count: number, word: string): string {
  return count === 1 ? `1 ${word}` : `${count} ${word}s`;
}

function createObjectURL(file: File): string {
  return URL.createObjectURL(file);
}

// ── component ─────────────────────────────────────────────────────────────────

export function GuestUploadClient({ event }: { event: EventProps }) {
  const [step, setStep] = useState<Step>('nickname');
  const [nickname, setNickname] = useState('');
  const [nicknameError, setNicknameError] = useState('');
  const [photos, setPhotos] = useState<PhotoState[]>([]);
  const [currentPhotoIndex, setCurrentPhotoIndex] = useState(0);
  const [uploadError, setUploadError] = useState('');
  const fileInputRef = useRef<HTMLInputElement>(null);

  // Thumbnails: object URLs keyed by file identity
  const thumbnailsRef = useRef<Map<File, string>>(new Map());

  function getThumbnail(file: File): string {
    if (!thumbnailsRef.current.has(file)) {
      thumbnailsRef.current.set(file, createObjectURL(file));
    }
    return thumbnailsRef.current.get(file)!;
  }

  // Cleanup object URLs on unmount
  useEffect(() => {
    return () => {
      thumbnailsRef.current.forEach((url) => URL.revokeObjectURL(url));
    };
  }, []);

  // beforeunload warning during upload
  useEffect(() => {
    if (step !== 'uploading') return;
    const handler = (e: BeforeUnloadEvent) => {
      e.preventDefault();
    };
    window.addEventListener('beforeunload', handler);
    return () => window.removeEventListener('beforeunload', handler);
  }, [step]);

  // ── Step 1: Nickname ────────────────────────────────────────────────────────

  function handleNicknameSubmit() {
    if (!nickname.trim()) {
      setNicknameError('Please enter your name to continue.');
      return;
    }
    setNicknameError('');
    setStep('select');
  }

  // ── Step 2: File selection ──────────────────────────────────────────────────

  function handleFilesSelected(e: React.ChangeEvent<HTMLInputElement>) {
    const files = Array.from(e.target.files ?? []);
    if (files.length === 0) return;
    setPhotos(files.map((file) => ({ file, status: 'pending', progress: 0 })));
  }

  const overLimit = photos.length > event.photoLimit;
  const excess = photos.length - event.photoLimit;

  // ── Step 3: Sequential XHR upload ─────────────────────────────────────────

  async function startUpload() {
    if (photos.length === 0 || overLimit) return;
    setStep('uploading');
    setCurrentPhotoIndex(0);

    let allFailed = true;
    let failedCount = 0;

    for (let i = 0; i < photos.length; i++) {
      setCurrentPhotoIndex(i);

      // Mark current photo as uploading
      setPhotos((prev) =>
        prev.map((p, idx) =>
          idx === i ? { ...p, status: 'uploading', progress: 0 } : p,
        ),
      );

      const success = await uploadSinglePhoto(photos[i].file, i);

      if (success) {
        allFailed = false;
        setPhotos((prev) =>
          prev.map((p, idx) =>
            idx === i ? { ...p, status: 'done', progress: 100 } : p,
          ),
        );
      } else {
        failedCount++;
        setPhotos((prev) =>
          prev.map((p, idx) =>
            idx === i ? { ...p, status: 'failed', progress: 0 } : p,
          ),
        );
      }
    }

    if (allFailed) {
      setUploadError('Your photos couldn\'t be uploaded. Check your connection and try again.');
      setStep('error_upload');
    } else if (failedCount > 0) {
      setUploadError(
        `${plural(failedCount, 'photo')} couldn't be uploaded. Check your connection and try again.`,
      );
      setStep('error_upload');
    } else {
      setStep('complete');
    }
  }

  function uploadSinglePhoto(file: File, photoIndex: number): Promise<boolean> {
    return new Promise((resolve) => {
      const formData = new FormData();
      formData.append('file', file);
      formData.append('nickname', nickname.trim());
      formData.append('photoIndex', String(photoIndex));

      const xhr = new XMLHttpRequest();
      xhr.open('POST', `/api/upload/${event.slug}`);

      xhr.upload.addEventListener('progress', (e) => {
        if (e.lengthComputable) {
          const pct = Math.round((e.loaded / e.total) * 100);
          setPhotos((prev) =>
            prev.map((p, idx) =>
              idx === photoIndex ? { ...p, progress: pct } : p,
            ),
          );
        }
      });

      xhr.addEventListener('load', () => {
        resolve(xhr.status >= 200 && xhr.status < 300);
      });

      xhr.addEventListener('error', () => resolve(false));
      xhr.addEventListener('abort', () => resolve(false));

      xhr.send(formData);
    });
  }

  // ── Retry: back to select ──────────────────────────────────────────────────

  function handleRetry() {
    setPhotos([]);
    setUploadError('');
    setStep('select');
    // Reset file input so onChange fires again if same files re-selected
    if (fileInputRef.current) fileInputRef.current.value = '';
  }

  // ── Event name header (Steps 1, 2, 3 only) ────────────────────────────────

  const showEventHeader = step === 'nickname' || step === 'select' || step === 'uploading';

  // ── Render ─────────────────────────────────────────────────────────────────

  return (
    <main className="min-h-screen flex flex-col items-center justify-center bg-[#faf9f7]">
      <div className="w-full max-w-sm px-6 py-12">

        {/* Event name header — Steps 1, 2, 3 */}
        {showEventHeader && (
          <div className="mb-0">
            <h1 className="text-3xl font-semibold text-center text-gray-800">
              {event.name}
            </h1>
            <div className="w-8 h-[2px] bg-[#b85c52] mx-auto mt-2 mb-8" />
          </div>
        )}

        {/* ── Step 1: Nickname ── */}
        {step === 'nickname' && (
          <div>
            <h2 className="text-3xl font-semibold mt-8">What&apos;s your name?</h2>
            <p className="text-base text-gray-500 mt-2 leading-relaxed">
              We&apos;ll use it to label your photos in the album.
            </p>
            <input
              type="text"
              autoComplete="nickname"
              autoCapitalize="words"
              autoCorrect="off"
              inputMode="text"
              placeholder="Your name or nickname"
              maxLength={60}
              value={nickname}
              onChange={(e) => setNickname(e.target.value)}
              onKeyDown={(e) => e.key === 'Enter' && handleNicknameSubmit()}
              className="w-full min-h-[52px] px-4 border border-gray-200 rounded-xl bg-white text-base font-normal mt-4 outline-none focus:ring-2 focus:ring-[#b85c52] focus:border-transparent"
            />
            {nicknameError && (
              <p className="text-sm text-red-600 mt-2">{nicknameError}</p>
            )}
            <button
              type="button"
              onClick={handleNicknameSubmit}
              disabled={!nickname.trim()}
              style={{ touchAction: 'manipulation' }}
              className="w-full min-h-[56px] rounded-xl mt-4 bg-[#b85c52] text-white font-semibold text-base disabled:opacity-50 disabled:cursor-not-allowed"
            >
              Continue →
            </button>
          </div>
        )}

        {/* ── Step 2: Photo selection ── */}
        {step === 'select' && (
          <div>
            <h2 className="text-3xl font-semibold mt-8">Choose your photos</h2>

            {/* Subtext — changes based on selection state */}
            {photos.length === 0 && (
              <p className="text-base text-gray-500 mt-2 leading-relaxed">
                Select up to {event.photoLimit} photos from your camera roll.
              </p>
            )}
            {photos.length > 0 && !overLimit && (
              <p className="text-sm font-semibold text-[#b85c52] mt-2">
                {photos.length} of {event.photoLimit} photos selected
              </p>
            )}
            {overLimit && (
              <div className="mt-2">
                <p className="text-sm font-semibold text-red-600">
                  {photos.length} of {event.photoLimit} photos selected — {excess} too many
                </p>
                <p className="text-sm text-red-600">
                  Remove {plural(excess, 'photo')} to continue.
                </p>
              </div>
            )}

            {/* Hidden file input */}
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*,.HEIC,.heic"
              multiple
              className="sr-only"
              onChange={handleFilesSelected}
            />

            {/* Photo list — shown when files selected */}
            {photos.length > 0 && (
              <div className="flex flex-col gap-2 mt-4 max-h-[220px] overflow-y-auto">
                {photos.map((p, i) => (
                  <div
                    key={i}
                    className="flex items-center gap-3 bg-white rounded-lg px-3 min-h-[44px] border border-gray-200"
                  >
                    <img
                      src={getThumbnail(p.file)}
                      alt=""
                      className="w-10 h-10 rounded object-cover flex-shrink-0"
                    />
                    <span className="text-sm truncate text-gray-700 flex-1">
                      {p.file.name}
                    </span>
                  </div>
                ))}
              </div>
            )}

            {/* Change photos link — shown when files already selected */}
            {photos.length > 0 && (
              <button
                type="button"
                onClick={() => fileInputRef.current?.click()}
                style={{ touchAction: 'manipulation' }}
                className="text-sm font-semibold underline text-gray-600 mt-3 block"
              >
                Change photos
              </button>
            )}

            {/* Primary CTA */}
            <button
              type="button"
              onClick={photos.length === 0 ? () => fileInputRef.current?.click() : startUpload}
              disabled={overLimit}
              style={{ touchAction: 'manipulation' }}
              className="w-full min-h-[56px] rounded-xl mt-4 bg-[#b85c52] text-white font-semibold text-base disabled:opacity-50 disabled:cursor-not-allowed"
            >
              {photos.length === 0
                ? 'Select photos'
                : `Upload ${plural(photos.length, 'photo')} →`}
            </button>
          </div>
        )}

        {/* ── Step 3: Uploading ── */}
        {step === 'uploading' && (
          <div>
            <h2 className="text-3xl font-semibold">Uploading…</h2>
            <p className="text-base text-gray-500 mt-2">Please keep this page open.</p>

            <p className="text-sm font-semibold text-gray-600 mt-6">
              Photo {Math.min(currentPhotoIndex + 1, photos.length)} of {photos.length}
            </p>

            <div className="flex flex-col gap-2 mt-3 max-h-[220px] overflow-y-auto">
              {photos.map((p, i) => {
                const isPending = p.status === 'pending';
                const isUploading = p.status === 'uploading';
                const isDone = p.status === 'done';
                const isFailed = p.status === 'failed';

                return (
                  <div
                    key={i}
                    className={`bg-white border border-gray-200 rounded-xl px-4 py-3 ${
                      isPending ? 'opacity-40' : ''
                    }`}
                  >
                    <div className="flex items-center gap-3">
                      <img
                        src={getThumbnail(p.file)}
                        alt=""
                        className="w-10 h-10 rounded object-cover flex-shrink-0"
                      />
                      <span className="text-sm truncate text-gray-700 flex-1">
                        {p.file.name}
                      </span>
                      {isDone && (
                        <span className="text-sm font-semibold text-[#b85c52] flex-shrink-0">
                          ✓
                        </span>
                      )}
                      {isFailed && (
                        <span className="text-sm font-semibold text-red-600 flex-shrink-0">
                          ×
                        </span>
                      )}
                      {isUploading && (
                        <span className="text-sm text-gray-500 flex-shrink-0">
                          {p.progress}%
                        </span>
                      )}
                    </div>

                    {/* Progress bar — shown only while uploading */}
                    {isUploading && (
                      <div className="mt-2 h-1 rounded-full bg-[#f0ebe9] overflow-hidden">
                        <div
                          className="h-full rounded-full bg-[#b85c52] transition-[width] duration-150 ease-out"
                          style={{ width: `${p.progress}%` }}
                        />
                      </div>
                    )}
                  </div>
                );
              })}
            </div>
          </div>
        )}

        {/* ── Step 4: Complete ── */}
        {step === 'complete' && (
          <div className="text-center">
            <p className="text-6xl mt-8">🎉</p>
            <h2 className="text-4xl font-semibold mt-4">You&apos;re all done!</h2>
            <p className="text-base text-center text-gray-600 mt-4 leading-relaxed">
              Your {plural(photos.filter((p) => p.status === 'done').length, 'photo')} have
              been added to the album. Thank you for sharing the moment!
            </p>
            <p className="text-sm text-center text-gray-400 mt-10">
              You can close this page.
            </p>
          </div>
        )}

        {/* ── Error: Upload failed ── */}
        {step === 'error_upload' && (
          <div className="text-center">
            <h2 className="text-3xl font-semibold mt-8">Something went wrong.</h2>
            <p className="text-base text-center text-gray-500 mt-3">{uploadError}</p>
            <button
              type="button"
              onClick={handleRetry}
              style={{ touchAction: 'manipulation' }}
              className="w-full min-h-[56px] bg-[#b85c52] text-white font-semibold rounded-xl mt-8"
            >
              Try again
            </button>
          </div>
        )}

      </div>
    </main>
  );
}
```

**Implementation notes for the executor:**
- The `plural()` helper must produce singular ("1 photo") when count === 1
