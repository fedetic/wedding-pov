**Step A — Create `src/components/events/QRModal.tsx`:**

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import QRCode from "qrcode";

type Props = {
  slug: string;
  eventName: string;
  onClose: () => void;
};

export function QRModal({ slug, eventName, onClose }: Props) {
  const [dataUrl, setDataUrl] = useState<string | null>(null);
  const [genError, setGenError] = useState(false);
  const backdropRef = useRef<HTMLDivElement>(null);

  // Generate QR on mount — must be client-side (window.location.origin)
  useEffect(() => {
    async function generate() {
      try {
        // CRITICAL: QR content must be absolute URL with protocol (not just /e/slug)
        // QR scanners require a full URL to know the host — relative paths won't work
        const url = `${window.location.origin}/e/${slug}`;
        const result = await QRCode.toDataURL(url, {
          errorCorrectionLevel: "M", // Level M: 15% error recovery — standard for print
          margin: 2,                 // 2-module quiet zone ≈ 16px at 256px width
          width: 256,                // 256×256px as per UI-SPEC
        });
        setDataUrl(result);
      } catch (e) {
        console.error("[QRModal] QR generation failed:", e);
        setGenError(true);
      }
    }
    generate();
  }, [slug]);

  // Close on Escape key
  useEffect(() => {
    function handleKey(e: KeyboardEvent) {
      if (e.key === "Escape") onClose();
    }
    document.addEventListener("keydown", handleKey);
    return () => document.removeEventListener("keydown", handleKey);
  }, [onClose]);

  // Download QR PNG
  function handleDownload() {
    if (!dataUrl) return;
    const link = document.createElement("a");
    link.href = dataUrl;
    link.download = `qr-${slug}.png`; // e.g. qr-our-wedding-Kj9mX2.png
    link.click();
  }

  // Close when clicking backdrop (but not the card itself)
  function handleBackdropClick(e: React.MouseEvent<HTMLDivElement>) {
    if (e.target === backdropRef.current) onClose();
  }

  return (
    <div
      ref={backdropRef}
      onClick={handleBackdropClick}
      className="fixed inset-0 bg-black/50 z-50 flex items-center justify-center"
      role="dialog"
      aria-modal="true"
      aria-label="QR code modal"
    >
      <div className="bg-white rounded-lg p-6 w-full max-w-sm mx-4 relative">
        {/* Close button */}
        <button
          type="button"
          onClick={onClose}
          aria-label="Close QR code modal"
          className="absolute top-4 right-4 text-gray-400 hover:text-gray-600 text-xl leading-none"
        >
          ×
        </button>

        {/* Header */}
        <h2 className="text-lg font-semibold mb-1">QR code</h2>
        <p className="text-sm text-gray-500 mb-4">{eventName}</p>

        {/* QR image area */}
        <div className="flex justify-center mb-6">
          {genError ? (
            <p className="text-sm text-red-600 text-center py-8">
              Could not generate QR code. Please reload and try again.
            </p>
          ) : dataUrl ? (
            <img
              src={dataUrl}
              width={256}
              height={256}
              alt={`QR code for ${eventName}`}
            />
          ) : (
            // Loading state — show a placeholder while generating
            <div
              className="bg-gray-100 rounded flex items-center justify-center"
              style={{ width: 256, height: 256 }}
            >
              <span className="text-sm text-gray-400">Generating…</span>
            </div>
          )}
        </div>

        {/* Download button */}
        <button
          type="button"
          onClick={handleDownload}
          disabled={!dataUrl || genError}
          className={`bg-black text-white rounded px-4 py-2 text-sm font-semibold w-full hover:bg-gray-800 ${
            !dataUrl || genError ? "opacity-50 cursor-not-allowed" : ""
          }`}
        >
          {dataUrl ? "Download QR code" : "Generating…"}
        </button>
      </div>
    </div>
  );
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → QR generation | `window.location.origin` used client-side — no server data; QR contains only public slug URL |
| QR URL content | Slug is intentionally public — guests need it to upload; content of QR is not sensitive |
| Download trigger | `document.createElement("a")` + `.click()` — standard browser download; no server involved |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-11 | Information Disclosure | Slug in QR code | accept | Slugs are intentionally public — the QR code's entire purpose is to share the slug URL with guests. The `/e/[slug]` page (Phase 3) will enforce active/inactive status server-side: inactive events return a rejection message, not a 404, so the slug's existence is not secret. |
| T-02-12 | Information Disclosure | QR data URL in browser memory | accept | `dataUrl` is a base64-encoded PNG of a public URL — no credentials, no tokens. Stored only in component state; cleared on modal close (component unmounts). |
| T-02-13 | Tampering | QR URL generated client-side | accept | `window.location.origin` cannot be forged by the organizer in a way that damages the system — worst case they generate a QR pointing to their own host, which is the intended behavior. Guests scan the code; the organizer controls the content. |
| T-02-14 | Denial of Service | QRCode.toDataURL in browser | accept | QR generation is CPU-bound but milliseconds at 256px; 1–2 organizer users; no server resources consumed; browser-side only. |
</threat_model>
