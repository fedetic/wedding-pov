From node_modules/better-auth/dist/api/routes/update-user.mjs (verified deleteUser behavior):
- Endpoint: POST `/delete-user`
- Body: `{ password?, callbackURL?, token? }`
- Middleware: `sensitiveSessionMiddleware` — rejects with `BASE_ERROR_CODES.SESSION_EXPIRED` if `Date.now() - session.createdAt >= freshAge` (default 86400 sec = 24h) AND no password provided
- On success: calls `internalAdapter.deleteUser(userId)` → DELETE FROM users → cascade → also deletes sessions, then deleteSessionCookie

DB cascade chain (from src/lib/db/schema.ts):
- `users.id` ← `sessions.userId` (onDelete: cascade)
- `users.id` ← `accounts.userId` (onDelete: cascade)
- `users.id` ← `googleTokens.userId` (onDelete: cascade)
- `users.id` ← `events.organizerId` (onDelete: cascade)
- `events.id` ← `upload_records.eventId` (onDelete: cascade)

Result: deleting a user removes the user, all sessions, all OAuth accounts, all encrypted Drive tokens, all events, and all upload records — verified by reading schema.ts.

Exact file content:

```tsx
"use client";

import { authClient } from "@/lib/auth-client";
import { useRouter } from "next/navigation";
import { useState } from "react";

type Status = "idle" | "confirming" | "deleting" | "stale" | "error";

export function DeleteAccountButton() {
  const router = useRouter();
  const [status, setStatus] = useState<Status>("idle");
  const [errorMessage, setErrorMessage] = useState<string>("");

  async function handleDelete() {
    setStatus("deleting");
    setErrorMessage("");
    const { data, error } = await authClient.deleteUser({});
    if (error) {
      // better-auth surfaces SESSION_EXPIRED via error.code or error.message.
      // freshAge default = 24h; user must sign out and sign back in.
      const code = error.code ?? "";
      const msg = error.message ?? "";
      if (code === "SESSION_EXPIRED" || /session.*expired/i.test(msg) || /fresh/i.test(msg)) {
        setStatus("stale");
        return;
      }
      setErrorMessage(msg || "Could not delete account. Please try again.");
      setStatus("error");
      return;
    }
    if (data?.success) {
      router.push("/login");
      return;
    }
    setErrorMessage("Unexpected response from server.");
    setStatus("error");
  }

  if (status === "stale") {
    return (
      <div className="flex flex-col gap-2">
        <p className="text-sm text-red-700">
          For security, please sign out and sign in again before deleting your account.
        </p>
        <button
          type="button"
          onClick={() => setStatus("idle")}
          className="text-sm text-gray-500 hover:text-black underline self-start"
        >
          Dismiss
        </button>
      </div>
    );
  }

  if (status === "error") {
    return (
      <div className="flex flex-col gap-2">
        <p className="text-sm text-red-700">{errorMessage}</p>
        <button
          type="button"
          onClick={() => setStatus("idle")}
          className="text-sm text-gray-500 hover:text-black underline self-start"
        >
          Dismiss
        </button>
      </div>
    );
  }

  if (status === "confirming" || status === "deleting") {
    return (
      <div className="flex flex-col gap-3">
        <p className="text-sm text-gray-700">
          This permanently deletes your account, Drive credentials, events, and all upload
          records. Photos already saved in your Google Drive remain in your Drive.
        </p>
        <div className="flex gap-3">
          <button
            type="button"
            onClick={handleDelete}
            disabled={status === "deleting"}
            className="text-sm text-red-600 hover:text-red-800 font-semibold disabled:opacity-50"
          >
            {status === "deleting" ? "Deleting…" : "Confirm delete"}
          </button>
          <button
            type="button"
            onClick={() => setStatus("idle")}
            disabled={status === "deleting"}
            className="text-sm text-gray-500 hover:text-gray-700 disabled:opacity-50"
          >
            Cancel
          </button>
        </div>
      </div>
    );
  }

  // idle state
  return (
    <button
      type="button"
      onClick={() => setStatus("confirming")}
      className="text-sm text-red-600 hover:text-red-800 underline self-start"
    >
      Delete account
    </button>
  );
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → /delete-user POST | Authenticated session crosses into a destructive operation |
| /delete-user → Postgres DELETE | Cascade deletes propagate across 5 tables |
| client → /login redirect | Post-deletion redirect must clear local session state |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-05 | Tampering | Attacker triggers /delete-user via CSRF | mitigate | better-auth's `sensitiveSessionMiddleware` requires fresh session AND same-origin cookies; Better Auth uses HTTP-only SameSite=Lax cookies |
| T-07-06 | Denial of service | Stale session lets organizer click but request fails | mitigate | UI explicitly handles SESSION_EXPIRED, shows actionable message ("sign out and sign in again") |
| T-07-07 | Information disclosure | Error message leaks session details | mitigate | DeleteAccountButton catches error.code/message and renders sanitized text; raw API error never reaches the DOM |
| T-07-08 | Repudiation | User clicks Delete by accident, no undo | mitigate | Two-step confirm (Delete account → Confirm delete) prevents single-tap mistakes; explanatory copy lists what gets deleted |
| T-07-09 | Elevation of privilege | Disabled deleteUser endpoint accessed | accept | Endpoint returns 404 when `user.deleteUser.enabled !== true`; Task 1 enables it explicitly — if someone disables it, the button will fail safely |
| T-07-SC | Tampering | npm install in this plan | accept | No package installs — N/A |
</threat_model>
